async (page, options = {}) => {
  const checks = [], errors = [];
  const verify = (value, message) => { if (!value) throw new Error(message); checks.push(message); };
  page.on('pageerror', error => errors.push(error.message));
  const jump = async id => {
    await page.locator('#jump-section').fill(String(id));
    await page.locator('#jump-form button').click();
    await page.waitForFunction(n => document.querySelector('#breadcrumb-section').textContent === `第 ${n} 节`, id);
  };
  await page.setViewportSize({width:1440,height:1000});
  await page.goto('http://127.0.0.1:4175/');
  await page.evaluate(() => document.fonts.ready);
  if (options.stage === 'chapters') {
    for (let id = options.from || 1; id <= (options.to || 100); id++) {
      await jump(id);
      const expected = await page.evaluate(n => window.ESSAYS[n].closeReading?.length ? 11 : 10, id);
      verify(await page.locator('.essay-section').count() === expected, `第 ${id} 节完整呈现`);
      verify(await page.locator('#article-toc a').count() === expected, `第 ${id} 节目录与正文对应`);
      verify((await page.locator('#text-content .prose').innerText()).length > 10, `第 ${id} 节中文自译可读`);
    }
    verify(errors.length === 0, '各节切换无运行异常');
    return {passed:checks.length,range:[options.from || 1,options.to || 100],errors};
  }
  await page.evaluate(() => localStorage.removeItem('pi-essay-reader-v2'));
  await page.reload();
  await page.evaluate(() => document.fonts.ready);
  verify(await page.evaluate(() => window.LESSONS.length === 100 && Object.keys(window.ORIGINALS).length === 100 && Object.keys(window.ESSAYS).length === 100), '百节数据加载完整');
  verify(await page.locator('#chapter-list .chapter').count() === 100, '百节目录齐全');
  verify(await page.locator('.chapter-group').count() === 10, '目录按十节分组');
  verify(await page.locator('textarea, .choice, #practice, #notes, #export, #complete, progress').count() === 0, '没有练习、留言、笔记和打卡');
  verify(await page.evaluate(() => [...document.fonts].some(font => font.family.includes('PI Reader WenKai') && font.status === 'loaded')), '霞鹜文楷实际加载');
  verify(await page.locator('.brand-icon').evaluate(node => getComputedStyle(node).backgroundColor === 'rgb(204, 125, 94)'), '主色为截图指定的 CC7D5E');
  await jump(67);
  verify(await page.locator('.chapter.active').getAttribute('href') === '#section-67', '节号跳转到第67节');
  verify(await page.locator('.chapter.active').evaluate(node => node.closest('details').open), '当前节所在分组自动展开');
  await page.locator('#article-toc a[href="#section-67/concepts"]').click();
  await page.reload();
  await page.evaluate(() => document.fonts.ready);
  verify((await page.title()).includes('第 67 节'), '分享小标题链接保留节号');
  await page.waitForTimeout(150);
  verify(await page.locator('#concepts').evaluate(node => node.getBoundingClientRect().top < 170), '小标题刷新后定位正文');
  await jump(100);
  verify(await page.locator('#next').isDisabled(), '第100节禁止越界');
  await page.locator('#previous').click();
  await page.waitForFunction(() => document.querySelector('#breadcrumb-section').textContent === '第 99 节');
  verify((await page.title()).includes('第 99 节'), '末尾上一节导航');
  await page.goBack();
  await page.waitForFunction(() => document.querySelector('#breadcrumb-section').textContent === '第 100 节');
  verify((await page.title()).includes('第 100 节'), '浏览器后退恢复节次');
  await page.locator('#search').fill('100');
  verify(await page.locator('#chapter-list .chapter').count() === 1, '数字搜索精确匹配节号');
  await page.locator('#search').fill('纤维');
  verify(await page.locator('#chapter-list a[href="#section-67"]').isVisible(), '精读正文可被全文搜索');
  await page.locator('#search').fill('xxxxxxxx无此内容');
  verify(await page.locator('#empty-search').isVisible(), '搜索空结果提示');
  await page.locator('#search').fill('');
  for (const id of [11,22,35,43,48,50,65,67,71,79,86,89,100]) {
    await jump(id);
    await page.locator('#tab-de').click();
    verify((await page.locator('#text-content .prose').innerText()).length > 20, `第 ${id} 节德文显示`);
    await page.locator('#tab-both').click();
    verify(await page.locator('#text-content .prose').count() === 2, `第 ${id} 节中德对照`);
    if (id === 48 || id === 86) {
      const images = page.locator('#text-content .prose[lang="zh-CN"] img');
      verify(await images.count() === 2, `第 ${id} 节中文保留两幅原图`);
      for (const image of await images.all()) {
        await image.scrollIntoViewIfNeeded();
        await image.evaluate(node => node.complete ? Promise.resolve() : new Promise(resolve => node.addEventListener('load', resolve, {once:true})));
        verify(await image.evaluate(node => node.naturalWidth > 0), `第 ${id} 节原图加载成功`);
      }
    }
    await page.locator('#tab-zh').click();
  }
  await jump(67);
  await page.evaluate(() => window.scrollTo({top:0,behavior:'instant'}));
  await page.screenshot({path:'output/playwright/desktop-100.png'});
  await page.locator('#theme').click();
  await page.locator('#font-size').click();
  await page.reload();
  verify(await page.locator('body').evaluate(node => node.classList.contains('dark') && node.classList.contains('large-type')), '主题与字号持久化');
  for (const width of [320,360,390,768,1024,1440]) {
    await page.setViewportSize({width,height:900});
    verify(await page.evaluate(() => document.documentElement.scrollWidth <= innerWidth), `${width}像素大字无横向溢出`);
  }
  await page.setViewportSize({width:390,height:844});
  await page.locator('#menu').click();
  verify(await page.locator('#sidebar').getAttribute('aria-modal') === 'true', '手机目录模态展开');
  await page.locator('#jump-section').fill('100');
  await page.locator('#jump-form button').click();
  await page.waitForFunction(() => document.querySelector('#breadcrumb-section').textContent === '第 100 节');
  verify(await page.locator('#sidebar').evaluate(node => node.inert), '手机跳转后关闭目录并释放焦点');
  verify(await page.locator('.workspace').evaluate(node => !node.inert), '正文恢复交互');
  await page.locator('#menu').click();
  await page.locator('#search').fill('家族相似');
  await page.locator('#chapter-list a[href="#section-67"]').click();
  await page.waitForTimeout(280);
  await page.locator('#theme').click();
  await page.locator('#font-size').click();
  await page.evaluate(() => window.scrollTo({top:0,behavior:'instant'}));
  await page.screenshot({path:'output/playwright/mobile-100.png'});
  await page.locator('#menu').click();
  await page.locator('#search').fill('');
  await page.waitForTimeout(250);
  await page.screenshot({path:'output/playwright/mobile-menu-100.png'});
  await page.keyboard.press('Escape');
  verify(await page.locator('#menu').evaluate(node => node === document.activeElement), 'Escape后焦点返回目录按钮');
  await page.locator('#menu').click();
  await page.locator('#about-open').click();
  verify(await page.locator('#about').isVisible(), '手机来源说明打开');
  await page.locator('#about-close').click();
  verify(!await page.locator('#about').isVisible(), '手机来源说明关闭');
  verify(errors.length === 0, '无浏览器运行错误');
  await page.evaluate(() => localStorage.removeItem('pi-essay-reader-v2'));
  return {passed:checks.length,errors};
}
