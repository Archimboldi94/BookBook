async (page, options = {}) => {
  const base = options.base || 'http://127.0.0.1:4176/philosophical-investigations/';
  const errors = [];
  page.on('pageerror', error => errors.push(error.message));
  await page.setViewportSize({width:1440,height:1000});
  await page.goto(base);
  await page.locator('#lesson h1').waitFor();
  await page.evaluate(() => document.fonts.ready);
  if (await page.locator('#tab-de,[lang=de],.tabs').count()) throw new Error('阅读页仍有德文入口');
  for (let id = 1; id <= 100; id++) {
    await page.locator('#jump-section').fill(String(id));
    await page.locator('#jump-form button').click();
    await page.waitForFunction(n => document.querySelector('#breadcrumb-section').textContent === `第 ${n} 节`, id);
    const expected = id > 10 ? 11 : 10;
    if (await page.locator('.essay-section').count() !== expected) throw new Error(`第 ${id} 节结构不完整`);
    const quotes = await page.locator('.close-reading-item blockquote').allTextContents();
    if (!quotes.every(quote => /[\u4e00-\u9fff]/.test(quote))) throw new Error(`第 ${id} 节精读仍有外文引句`);
    if (!(await page.locator('#text-content').innerText()).trim()) throw new Error(`第 ${id} 节中文文本为空`);
  }
  if (!await page.locator('#next').isDisabled()) throw new Error('第 100 节存在越界导航');
  const font = await page.evaluate(() => [...document.fonts].some(item => item.family.includes('PI Reader WenKai') && item.status === 'loaded'));
  if (!font) throw new Error('霞鹜文楷未加载');
  const color = await page.locator('.brand-icon').evaluate(node => getComputedStyle(node).backgroundColor);
  if (color !== 'rgb(204, 125, 94)') throw new Error('主色不匹配');
  for (const width of [320,390,768,1440]) {
    await page.setViewportSize({width,height:900});
    if (!await page.evaluate(() => document.documentElement.scrollWidth <= innerWidth)) throw new Error(`${width} 像素横向溢出`);
  }
  if (errors.length) throw new Error(errors.join(';'));
  return {chapters:100,chineseReading:true,fontLoaded:true,color,errors};
}
